Segment Tree

What it is

Binary tree over array intervals. Supports range query and point/range update in O(log n) per operation.

When to use

  • Range sum / min / max / gcd.
  • Range assign or lazy propagation.
  • When BIT is not enough (non-invertible operations).

Lazy propagation template (range add, range sum)

struct SegTree {
    int n;
    vector<long long> st, lazy;
    SegTree(int n) : n(n), st(4*n), lazy(4*n) {}
    void push(int p, int l, int r) {
        if (!lazy[p]) return;
        st[p] += lazy[p] * (r - l + 1);
        if (l != r) { lazy[2*p] += lazy[p]; lazy[2*p+1] += lazy[p]; }
        lazy[p] = 0;
    }
    void upd(int p, int l, int r, int ql, int qr, long long v) {
        push(p, l, r);
        if (qr < l || r < ql) return;
        if (ql <= l && r <= qr) { lazy[p] += v; push(p, l, r); return; }
        int m = (l + r) / 2;
        upd(2*p, l, m, ql, qr, v);
        upd(2*p+1, m+1, r, ql, qr, v);
        push(2*p, l, m); push(2*p+1, m+1, r);
        st[p] = st[2*p] + st[2*p+1];
    }
    long long query(int p, int l, int r, int ql, int qr) {
        push(p, l, r);
        if (qr < l || r < ql) return 0;
        if (ql <= l && r <= qr) return st[p];
        int m = (l + r) / 2;
        return query(2*p, l, m, ql, qr) + query(2*p+1, m+1, r, ql, qr);
    }
};
