Matrix Exponentiation

What it is

Multiply n×n transition matrix to power k in O(n³ log k) to solve linear recurrences for huge k.

When to use

  • Fibonacci-like: F(n) = c1*F(n-1) + c2*F(n-2).
  • Count paths of length exactly k in graph (adjacency matrix power).
  • Linear DP where state is small vector.

Template

using Matrix = vector<vector<long long>>;

Matrix mul(const Matrix& A, const Matrix& B, long long mod) {
    int n = A.size();
    Matrix C(n, vector<long long>(n, 0));
    for (int i = 0; i < n; i++)
        for (int k = 0; k < n; k++) if (A[i][k])
            for (int j = 0; j < n; j++)
                C[i][j] = (C[i][j] + A[i][k] * B[k][j]) % mod;
    return C;
}

Matrix power(Matrix A, long long e, long long mod) {
    int n = A.size();
    Matrix res(n, vector<long long>(n, 0));
    for (int i = 0; i < n; i++) res[i][i] = 1;
    while (e) {
        if (e & 1) res = mul(res, A, mod);
        A = mul(A, A, mod);
        e >>= 1;
    }
    return res;
}
