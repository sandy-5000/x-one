DSU on Tree (Small-to-Large)

What it is

Process tree nodes in DFS order; when leaving a node, keep only heavy child's frequency map and merge light children. Subtree queries in O(n log n).

When to use

  • Count distinct colors in subtree.
  • Mode / majority in subtree.
  • Offline subtree frequency queries.

Template sketch

int heavy[MAXN], sz[MAXN];
map<int,int> cnt;
int cur_ans;

void add(int u) { /* add node u to cnt, update cur_ans */ }
void remove(int u) { /* undo */ }
void clear_all() { cnt.clear(); cur_ans = 0; }

int dfs_sz(int u, int p) {
    sz[u] = 1; heavy[u] = -1;
    for (int v : adj[u]) if (v != p) {
        int s = dfs_sz(v, u);
        sz[u] += s;
        if (heavy[u] == -1 || s > sz[heavy[u]]) heavy[u] = v;
    }
    return sz[u];
}

void dfs(int u, int p, bool keep) {
    for (int v : adj[u]) if (v != p && v != heavy[u]) dfs(v, u, false);
    if (heavy[u] != -1) dfs(heavy[u], u, true);
    for (int v : adj[u]) if (v != p && v != heavy[u])
        for (int x : nodes_in_subtree(v)) add(x);
    add(u);
    answer[u] = cur_ans;
    if (!keep) clear_all();
}
