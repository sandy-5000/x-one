Topological Sort

What it is

Linear ordering of DAG vertices such that for every edge u → v, u comes before v. Detects cycle if impossible.

When to use

  • Task dependencies.
  • DP on DAG (process in topo order).
  • Course prerequisites.

Kahn's algorithm (BFS)

vector<int> indeg(n);
for (auto& e : edges) indeg[e.to]++;
queue<int> q;
for (int i = 0; i < n; i++) if (!indeg[i]) q.push(i);
vector<int> order;
while (!q.empty()) {
    int u = q.front(); q.pop();
    order.push_back(u);
    for (int v : adj[u]) if (--indeg[v] == 0) q.push(v);
}
bool is_dag = order.size() == n;

DFS version

vector<int> order, state(n); // 0 unvisited, 1 in stack, 2 done
bool cycle = false;
function<void(int)> dfs = [&](int u) {
    state[u] = 1;
    for (int v : adj[u]) {
        if (state[v] == 1) cycle = true;
        if (!state[v]) dfs(v);
    }
    state[u] = 2;
    order.push_back(u);
};
// reverse order for topo (or use order as postorder reversed)
