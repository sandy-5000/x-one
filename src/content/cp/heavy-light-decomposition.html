Heavy-Light Decomposition (HLD)

What it is

Decompose tree into heavy paths (chains). Path queries become O(log² n) segment tree queries on chains.

When to use

  • Path sum/min/max/update on tree.
  • Path edge queries with segment tree on depth order.
  • Common in Div1 A/B and hard Div2 D.

Key ideas

  1. Heavy child: child with largest subtree size.
  2. Head of chain: top of heavy path.
  3. Query path: jump from chain head to LCA level by level.

Template sketch

int parent[MAXN], depth[MAXN], heavy[MAXN], head[MAXN], pos[MAXN];
int cur_pos = 0;

int dfs(int u, int p) {
    int size = 1, max_sz = 0;
    for (int v : adj[u]) if (v != p) {
        parent[v] = u; depth[v] = depth[u] + 1;
        int s = dfs(v, u);
        size += s;
        if (s > max_sz) { max_sz = s; heavy[u] = v; }
    }
    return size;
}

void decompose(int u, int h) {
    head[u] = h;
    pos[u] = cur_pos++;
    if (heavy[u]) decompose(heavy[u], h);
    for (int v : adj[u]) if (v != parent[u] && v != heavy[u])
        decompose(v, v);
}

long long query_path(int u, int v) {
    long long res = 0;
    for (; head[u] != head[v]; v = parent[head[v]]) {
        if (depth[head[u]] < depth[head[v]]) swap(u, v);
        res += seg.query(pos[head[u]], pos[u]);
    }
    if (depth[u] > depth[v]) swap(u, v);
    res += seg.query(pos[u], pos[v]);
    return res;
}
