LCA (Binary Lifting)

What it is

Lowest Common Ancestor of two nodes in tree. Binary lifting preprocesses ancestors at powers of 2 for O(log n) per query.

When to use

  • Distance between two nodes: dist(u,v) = depth[u] + depth[v] - 2*depth[lca].
  • Path queries (sometimes with segment tree on Euler tour).
  • K-th ancestor queries.

Preprocess template

const int LOG = 20;
vector<vector<int>> up(n, vector<int>(LOG));
vector<int> depth;

void dfs(int u, int p) {
    up[u][0] = p;
    for (int k = 1; k < LOG; k++)
        up[u][k] = up[up[u][k-1]][k-1];
    for (int v : adj[u]) if (v != p) {
        depth[v] = depth[u] + 1;
        dfs(v, u);
    }
}

int lca(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    int diff = depth[u] - depth[v];
    for (int k = 0; k < LOG; k++)
        if (diff >> k & 1) u = up[u][k];
    if (u == v) return u;
    for (int k = LOG-1; k >= 0; k--)
        if (up[u][k] != up[v][k]) { u = up[u][k]; v = up[v][k]; }
    return up[u][0];
}

int kth_ancestor(int u, int k) {
    for (int i = 0; i < LOG; i++)
        if (k >> i & 1) u = up[u][i];
    return u;
}
